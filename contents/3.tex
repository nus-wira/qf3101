\[N-d_{0,K}N=N(1-d_{0,K})\]
\[r = \frac{1-d_{0,K}}{\sum_{i=1}^K \frac{t_i}{360}d_{0,i}}\]

\subsection*{3.2.2 Valuation of Swap}
\[B_{fl} = (N+f_1)\exp{(-r_1 t_1)}\]
\[B_{fix} = N\exp{(-r_n t_n)} + \sum_{i=1}^n k\exp{(-r_i t_i)}\]
\[V=B_{fix}-B_{fl}\]
